SET NAMES utf8mb4;

CREATE DATABASE IF NOT EXISTS comunigest_db
    CHARACTER SET utf8mb4
    COLLATE utf8mb4_unicode_ci;

-- Usuario utilizado por PHP dentro de Docker.
-- Estas instrucciones deben ejecutarse con un usuario administrador.
CREATE USER IF NOT EXISTS 'appuser'@'%' IDENTIFIED BY 'apppass';
GRANT SELECT, INSERT, UPDATE, DELETE
    ON comunigest_db.*
    TO 'appuser'@'%';

USE comunigest_db;

CREATE TABLE IF NOT EXISTS comunidades (
    id_comunidad INT UNSIGNED AUTO_INCREMENT,
    nombre VARCHAR(120) NOT NULL,
    descripcion TEXT NULL,
    estado ENUM('Activa', 'Inactiva') NOT NULL DEFAULT 'Activa',
    fecha_creacion TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    fecha_actualizacion TIMESTAMP NOT NULL
        DEFAULT CURRENT_TIMESTAMP
        ON UPDATE CURRENT_TIMESTAMP,

    CONSTRAINT pk_comunidades
        PRIMARY KEY (id_comunidad),
    CONSTRAINT uq_comunidades_nombre
        UNIQUE (nombre)
) ENGINE = InnoDB
  COMMENT = 'Asociaciones o comunidades administradas por ComuniGest';

CREATE TABLE IF NOT EXISTS roles (
    id_rol TINYINT UNSIGNED AUTO_INCREMENT,
    nombre VARCHAR(50) NOT NULL,
    descripcion VARCHAR(255) NULL,
    estado ENUM('Activo', 'Inactivo') NOT NULL DEFAULT 'Activo',
    fecha_creacion TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    fecha_actualizacion TIMESTAMP NOT NULL
        DEFAULT CURRENT_TIMESTAMP
        ON UPDATE CURRENT_TIMESTAMP,

    CONSTRAINT pk_roles
        PRIMARY KEY (id_rol),
    CONSTRAINT uq_roles_nombre
        UNIQUE (nombre)
) ENGINE = InnoDB
  COMMENT = 'Perfiles de acceso disponibles dentro de una comunidad';

CREATE TABLE IF NOT EXISTS usuarios (
    id_usuario INT UNSIGNED AUTO_INCREMENT,
    nombre VARCHAR(120) NOT NULL,
    correo VARCHAR(191) NOT NULL,
    contrasena_hash VARCHAR(255) NOT NULL,
    telefono VARCHAR(20) NULL,
    estado ENUM('Pendiente', 'Activo', 'Inactivo', 'Bloqueado')
        NOT NULL DEFAULT 'Activo',
    ultimo_acceso DATETIME NULL,
    fecha_creacion TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    fecha_actualizacion TIMESTAMP NOT NULL
        DEFAULT CURRENT_TIMESTAMP
        ON UPDATE CURRENT_TIMESTAMP,

    CONSTRAINT pk_usuarios
        PRIMARY KEY (id_usuario),
    CONSTRAINT uq_usuarios_correo
        UNIQUE (correo),
    CONSTRAINT chk_usuarios_nombre
        CHECK (CHAR_LENGTH(TRIM(nombre)) >= 3)
) ENGINE = InnoDB
  COMMENT = 'Cuentas globales utilizadas para autenticarse';


CREATE TABLE IF NOT EXISTS usuario_comunidad (
    id_comunidad INT UNSIGNED NOT NULL,
    id_usuario INT UNSIGNED NOT NULL,
    id_rol TINYINT UNSIGNED NOT NULL,
    fecha_ingreso DATE NOT NULL,
    estado ENUM('Activo', 'Inactivo') NOT NULL DEFAULT 'Activo',
    fecha_creacion TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    fecha_actualizacion TIMESTAMP NOT NULL
        DEFAULT CURRENT_TIMESTAMP
        ON UPDATE CURRENT_TIMESTAMP,

    CONSTRAINT pk_usuario_comunidad
        PRIMARY KEY (id_comunidad, id_usuario),
    CONSTRAINT fk_uc_comunidad
        FOREIGN KEY (id_comunidad)
        REFERENCES comunidades (id_comunidad)
        ON UPDATE CASCADE
        ON DELETE RESTRICT,
    CONSTRAINT fk_uc_usuario
        FOREIGN KEY (id_usuario)
        REFERENCES usuarios (id_usuario)
        ON UPDATE CASCADE
        ON DELETE RESTRICT,
    CONSTRAINT fk_uc_rol
        FOREIGN KEY (id_rol)
        REFERENCES roles (id_rol)
        ON UPDATE CASCADE
        ON DELETE RESTRICT,

    INDEX idx_uc_usuario (id_usuario),
    INDEX idx_uc_rol (id_rol),
    INDEX idx_uc_estado (estado)
) ENGINE = InnoDB
  COMMENT = 'Membresia y rol de cada usuario en cada comunidad';


CREATE TABLE IF NOT EXISTS grupos (
    id_grupo INT UNSIGNED AUTO_INCREMENT,
    id_comunidad INT UNSIGNED NOT NULL,
    id_responsable INT UNSIGNED NOT NULL,
    nombre VARCHAR(120) NOT NULL,
    area VARCHAR(80) NOT NULL,
    descripcion TEXT NULL,
    estado ENUM('Activo', 'Inactivo') NOT NULL DEFAULT 'Activo',
    fecha_creacion TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    fecha_actualizacion TIMESTAMP NOT NULL
        DEFAULT CURRENT_TIMESTAMP
        ON UPDATE CURRENT_TIMESTAMP,

    CONSTRAINT pk_grupos
        PRIMARY KEY (id_grupo),
    CONSTRAINT uq_grupos_comunidad_nombre
        UNIQUE (id_comunidad, nombre),
    CONSTRAINT fk_grupos_comunidad
        FOREIGN KEY (id_comunidad)
        REFERENCES comunidades (id_comunidad)
        ON UPDATE CASCADE
        ON DELETE RESTRICT,
    CONSTRAINT fk_grupos_responsable
        FOREIGN KEY (id_responsable)
        REFERENCES usuarios (id_usuario)
        ON UPDATE CASCADE
        ON DELETE RESTRICT,

    INDEX idx_grupos_responsable (id_responsable),
    INDEX idx_grupos_estado (estado)
) ENGINE = InnoDB
  COMMENT = 'Comites y equipos de trabajo de cada comunidad';


CREATE TABLE IF NOT EXISTS usuario_grupo (
    id_grupo INT UNSIGNED NOT NULL,
    id_usuario INT UNSIGNED NOT NULL,
    rol_grupo VARCHAR(50) NOT NULL DEFAULT 'Colaborador',
    fecha_asignacion DATE NOT NULL,
    estado ENUM('Activo', 'Inactivo') NOT NULL DEFAULT 'Activo',
    fecha_creacion TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    fecha_actualizacion TIMESTAMP NOT NULL
        DEFAULT CURRENT_TIMESTAMP
        ON UPDATE CURRENT_TIMESTAMP,

    CONSTRAINT pk_usuario_grupo
        PRIMARY KEY (id_grupo, id_usuario),
    CONSTRAINT fk_ug_grupo
        FOREIGN KEY (id_grupo)
        REFERENCES grupos (id_grupo)
        ON UPDATE CASCADE
        ON DELETE CASCADE,
    CONSTRAINT fk_ug_usuario
        FOREIGN KEY (id_usuario)
        REFERENCES usuarios (id_usuario)
        ON UPDATE CASCADE
        ON DELETE RESTRICT,

    INDEX idx_ug_usuario (id_usuario),
    INDEX idx_ug_estado (estado)
) ENGINE = InnoDB
  COMMENT = 'Miembros asignados a grupos y su funcion interna';

CREATE TABLE IF NOT EXISTS proyectos (
    id_proyecto INT UNSIGNED AUTO_INCREMENT,
    id_grupo INT UNSIGNED NOT NULL,
    id_responsable INT UNSIGNED NOT NULL,
    nombre VARCHAR(150) NOT NULL,
    descripcion TEXT NULL,
    fecha_inicio DATE NOT NULL,
    fecha_fin DATE NULL,
    estado ENUM(
        'Planificado',
        'En proceso',
        'Pausado',
        'Finalizado',
        'Cancelado'
    ) NOT NULL DEFAULT 'Planificado',
    presupuesto DECIMAL(14, 2) NOT NULL DEFAULT 0.00,
    fecha_creacion TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    fecha_actualizacion TIMESTAMP NOT NULL
        DEFAULT CURRENT_TIMESTAMP
        ON UPDATE CURRENT_TIMESTAMP,

    CONSTRAINT pk_proyectos
        PRIMARY KEY (id_proyecto),
    CONSTRAINT uq_proyectos_grupo_nombre
        UNIQUE (id_grupo, nombre),
    CONSTRAINT fk_proyectos_grupo
        FOREIGN KEY (id_grupo)
        REFERENCES grupos (id_grupo)
        ON UPDATE CASCADE
        ON DELETE RESTRICT,
    CONSTRAINT fk_proyectos_responsable
        FOREIGN KEY (id_responsable)
        REFERENCES usuarios (id_usuario)
        ON UPDATE CASCADE
        ON DELETE RESTRICT,
    CONSTRAINT chk_proyectos_fechas
        CHECK (fecha_fin IS NULL OR fecha_fin >= fecha_inicio),
    CONSTRAINT chk_proyectos_presupuesto
        CHECK (presupuesto >= 0),

    INDEX idx_proyectos_responsable (id_responsable),
    INDEX idx_proyectos_estado (estado),
    INDEX idx_proyectos_fechas (fecha_inicio, fecha_fin)
) ENGINE = InnoDB
  COMMENT = 'Proyectos comunitarios administrados por los grupos';

CREATE TABLE IF NOT EXISTS tareas (
    id_tarea INT UNSIGNED AUTO_INCREMENT,
    id_proyecto INT UNSIGNED NOT NULL,
    id_responsable INT UNSIGNED NOT NULL,
    titulo VARCHAR(150) NOT NULL,
    descripcion TEXT NULL,
    fecha_inicio DATE NOT NULL,
    fecha_fin DATE NULL,
    estado ENUM('Pendiente', 'En proceso', 'Finalizada', 'Cancelada')
        NOT NULL DEFAULT 'Pendiente',
    fecha_creacion TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    fecha_actualizacion TIMESTAMP NOT NULL
        DEFAULT CURRENT_TIMESTAMP
        ON UPDATE CURRENT_TIMESTAMP,

    CONSTRAINT pk_tareas
        PRIMARY KEY (id_tarea),
    CONSTRAINT fk_tareas_proyecto
        FOREIGN KEY (id_proyecto)
        REFERENCES proyectos (id_proyecto)
        ON UPDATE CASCADE
        ON DELETE RESTRICT,
    CONSTRAINT fk_tareas_responsable
        FOREIGN KEY (id_responsable)
        REFERENCES usuarios (id_usuario)
        ON UPDATE CASCADE
        ON DELETE RESTRICT,
    CONSTRAINT chk_tareas_fechas
        CHECK (fecha_fin IS NULL OR fecha_fin >= fecha_inicio),

    INDEX idx_tareas_proyecto_estado (id_proyecto, estado),
    INDEX idx_tareas_responsable (id_responsable),
    INDEX idx_tareas_fechas (fecha_inicio, fecha_fin)
) ENGINE = InnoDB
  COMMENT = 'Trabajo necesario para completar cada proyecto';


CREATE TABLE IF NOT EXISTS actividades (
    id_actividad INT UNSIGNED AUTO_INCREMENT,
    id_comunidad INT UNSIGNED NOT NULL,
    id_proyecto INT UNSIGNED NULL,
    id_responsable INT UNSIGNED NOT NULL,
    titulo VARCHAR(150) NOT NULL,
    tipo ENUM('Reunion', 'Evento', 'Jornada', 'Proyecto') NOT NULL,
    descripcion TEXT NULL,
    fecha DATE NOT NULL,
    hora TIME NOT NULL,
    lugar VARCHAR(150) NOT NULL,
    estado ENUM('Programada', 'Realizada', 'Cancelada')
        NOT NULL DEFAULT 'Programada',
    fecha_creacion TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    fecha_actualizacion TIMESTAMP NOT NULL
        DEFAULT CURRENT_TIMESTAMP
        ON UPDATE CURRENT_TIMESTAMP,

    CONSTRAINT pk_actividades
        PRIMARY KEY (id_actividad),
    CONSTRAINT fk_actividades_comunidad
        FOREIGN KEY (id_comunidad)
        REFERENCES comunidades (id_comunidad)
        ON UPDATE CASCADE
        ON DELETE RESTRICT,
    CONSTRAINT fk_actividades_proyecto
        FOREIGN KEY (id_proyecto)
        REFERENCES proyectos (id_proyecto)
        ON UPDATE CASCADE
        ON DELETE RESTRICT,
    CONSTRAINT fk_actividades_responsable
        FOREIGN KEY (id_responsable)
        REFERENCES usuarios (id_usuario)
        ON UPDATE CASCADE
        ON DELETE RESTRICT,

    INDEX idx_actividades_comunidad_fecha (id_comunidad, fecha),
    INDEX idx_actividades_proyecto (id_proyecto),
    INDEX idx_actividades_responsable (id_responsable),
    INDEX idx_actividades_estado (estado)
) ENGINE = InnoDB
  COMMENT = 'Reuniones, eventos y jornadas de una comunidad';


CREATE TABLE IF NOT EXISTS actividad_participacion (
    id_actividad INT UNSIGNED NOT NULL,
    id_usuario INT UNSIGNED NOT NULL,
    estado ENUM('Confirmado', 'Asistio', 'Ausente', 'Cancelado')
        NOT NULL DEFAULT 'Confirmado',
    fecha_registro DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    observaciones VARCHAR(255) NULL,
    fecha_actualizacion TIMESTAMP NOT NULL
        DEFAULT CURRENT_TIMESTAMP
        ON UPDATE CURRENT_TIMESTAMP,

    CONSTRAINT pk_actividad_participacion
        PRIMARY KEY (id_actividad, id_usuario),
    CONSTRAINT fk_ap_actividad
        FOREIGN KEY (id_actividad)
        REFERENCES actividades (id_actividad)
        ON UPDATE CASCADE
        ON DELETE CASCADE,
    CONSTRAINT fk_ap_usuario
        FOREIGN KEY (id_usuario)
        REFERENCES usuarios (id_usuario)
        ON UPDATE CASCADE
        ON DELETE RESTRICT,

    INDEX idx_ap_usuario (id_usuario),
    INDEX idx_ap_estado (estado)
) ENGINE = InnoDB
  COMMENT = 'Confirmacion y asistencia de usuarios a actividades';


CREATE TABLE IF NOT EXISTS movimientos_financieros (
    id_movimiento INT UNSIGNED AUTO_INCREMENT,
    id_comunidad INT UNSIGNED NOT NULL,
    id_proyecto INT UNSIGNED NULL,
    id_usuario INT UNSIGNED NOT NULL,
    tipo ENUM('Ingreso', 'Egreso') NOT NULL,
    descripcion VARCHAR(255) NOT NULL,
    monto DECIMAL(14, 2) NOT NULL,
    fecha DATE NOT NULL,
    estado ENUM('Registrado', 'Anulado') NOT NULL DEFAULT 'Registrado',
    fecha_creacion TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    fecha_actualizacion TIMESTAMP NOT NULL
        DEFAULT CURRENT_TIMESTAMP
        ON UPDATE CURRENT_TIMESTAMP,

    CONSTRAINT pk_movimientos_financieros
        PRIMARY KEY (id_movimiento),
    CONSTRAINT fk_mf_comunidad
        FOREIGN KEY (id_comunidad)
        REFERENCES comunidades (id_comunidad)
        ON UPDATE CASCADE
        ON DELETE RESTRICT,
    CONSTRAINT fk_mf_proyecto
        FOREIGN KEY (id_proyecto)
        REFERENCES proyectos (id_proyecto)
        ON UPDATE CASCADE
        ON DELETE RESTRICT,
    CONSTRAINT fk_mf_usuario
        FOREIGN KEY (id_usuario)
        REFERENCES usuarios (id_usuario)
        ON UPDATE CASCADE
        ON DELETE RESTRICT,
    CONSTRAINT chk_mf_monto
        CHECK (monto > 0),

    INDEX idx_mf_comunidad_fecha (id_comunidad, fecha),
    INDEX idx_mf_proyecto (id_proyecto),
    INDEX idx_mf_usuario (id_usuario),
    INDEX idx_mf_tipo_fecha (tipo, fecha),
    INDEX idx_mf_estado (estado)
) ENGINE = InnoDB
  COMMENT = 'Ingresos y egresos generales o vinculados a proyectos';
